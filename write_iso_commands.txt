---> check what drives are connected to the pc and if in this case the usb is "sda" do the following

lsblk
sudo umount /dev/sda#   -->  unmout if the partition is mounted
sudo dd bs=128M if=/path/to/archlinux.iso of=/dev/sda conv=fsync oflag=direct status=progress

---> bzcat to unzip the compressed file and pipe directly to sdX usb
bzcat steamdeck-recovery-4.img.bz2 | sudo dd if=/dev/stdin of=/dev/sdX oflag=sync status=progress bs=128M
